  /* ─────────────────────────────────────────────────────────

  var toastTimer;
  function showToast(msg, type, duration) {
    var el = document.getElementById("toast");
    el.textContent = msg;
    el.className = "toast show" + (type ? " " + type : "");
    clearTimeout(toastTimer);
    toastTimer = setTimeout(function () { el.className = "toast"; }, duration || 3000);
  }

  function setSaveStatus(msg, type) {
    var el = document.getElementById("saveStatus");
    if (!el) return;
    el.textContent = msg;
    el.className = "save-status " + (type || "");
  }

  function setStatus(id, msg, type) {
    var el = document.getElementById(id);
    if (!el) return;
    el.textContent = msg;
    el.className = (el.className.replace(/\b(ok|err|info)\b/g, "")).trim() + (type ? " " + type : "");
  }

  function esc(str) {
    return String(str || "")
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;");
  }

  function fmtUsDate(isoOrSerial) {
    if (!isoOrSerial) return "";
    var s = String(isoOrSerial);
    if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(s)) return s;
    if (/^\d{4}-\d{2}-\d{2}/.test(s)) {
      var p = s.split("T")[0].split("-");
      return p[1] + "/" + p[2] + "/" + p[0];
    }
    if (/^\d+$/.test(s)) {
      var epoch = (Number(s) - 25569) * 86400000;
      var d = new Date(epoch);
      return (d.getUTCMonth()+1) + "/" + d.getUTCDate() + "/" + d.getUTCFullYear();
    }
    return s;
  }

  function todayUsDate() {
    var d = new Date();
    return (d.getMonth()+1) + "/" + d.getDate() + "/" + d.getFullYear();
  }

  function isoToUs(iso) {
    if (!iso) return "";
    var p = iso.split("-");
    return p[1] + "/" + p[2] + "/" + p[0];
  }

  function usToIso(us) {
    if (!us || us === "N/A") return "";
    var p = us.split("/");
    if (p.length !== 3) return "";
    return p[2] + "-" + p[0].padStart(2,"0") + "-" + p[1].padStart(2,"0");
  }

  function fmtCurrency(v) {
    var n = parseFloat(String(v).replace(/[^0-9.-]/g, ""));
    if (isNaN(n)) return String(v || "");
    return "$" + n.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  }

  /* ─────────────────────────────────────────────────────────
     Date picker wiring
  ───────────────────────────────────────────────────────── */
  function wireAllDatePickers() {
    document.querySelectorAll(".date-picker-button").forEach(function (btn) {
      var fieldName = btn.getAttribute("data-date-for");
      if (!fieldName) return;
      btn.addEventListener("click", function () {
        var native = document.querySelector('[data-date-picker="' + fieldName + '"]');
        if (native && !native.disabled) native.click();
      });
    });
    document.querySelectorAll(".native-date-picker").forEach(function (native) {
      var fieldName = native.getAttribute("data-date-picker");
      if (!fieldName) return;
      native.addEventListener("change", function () {
        var display = document.querySelector('[name="' + fieldName + '"]');
        if (display) display.value = isoToUs(native.value);
        // special: followupDate hidden field sync
        if (fieldName === "followupDate") {
          document.querySelector('[name="followupDate"]').value = isoToUs(native.value);
        }
      });
    });
  }

  /* ─────────────────────────────────────────────────────────
     Client lookup
  ───────────────────────────────────────────────────────── */
  var lookupDebounce;

  function initClientLookup() {
    var input = document.querySelector('[name="clientLookupQuery"]');
    if (!input) return;
    input.addEventListener("input", function () {
      clearTimeout(lookupDebounce);
      var q = input.value.trim();
      if (!q) return;
      lookupDebounce = setTimeout(function () { doClientLookup(q); }, 500);
    });
  }

  function doClientLookup(query) {
    setStatus("clientLookupError", "Looking up client…");
    google.script.run
      .withSuccessHandler(function (result) {
        if (!result || !result.rows || !result.rows.length) {
          setStatus("clientLookupError", "No client found.", "err");
          return;
        }
        applyClientData(result.rows[0]);
        if (result.warning) setStatus("clientLookupError", result.warning);
        else setStatus("clientLookupError", "");
      })
      .withFailureHandler(function (err) {
        setStatus("clientLookupError", err.message, "err");
      })
      .lookupClientReference(query);
  }

  function applyClientData(row) {
    var fields = {
      subscriber:           row.subscriber           || row.Subscriber          || "",
      clientId:             row.clientId             || row["Client ID"]        || "",
      clientFullName:       row.clientFullName       || row["Client Full Name"] || "",
      creditorName:         row.creditorName         || row["CREDITOR_NAME"]    || row["Creditor Name"] || "",
      currentAccountNumbers: row.currentAccountNumbers || row["DEBT_ACCOUNT_NUMBER"] || "",
      originalAccountNumber: row.originalAccountNumber || row["ORIGINAL_ACCOUNT_NUMBER"] || ""
    };
    Object.keys(fields).forEach(function (name) {
      var el = document.querySelector('[name="' + name + '"]');
      if (el && fields[name]) el.value = fields[name];
    });
  }

  /* ─────────────────────────────────────────────────────────
     CRM Schedule IDs (bulk paste)
  ───────────────────────────────────────────────────────── */
  var scheduledIds = [];

  function initBulkCrmIds() {
    var preview = document.getElementById("bulkCrmPreview");
    var applyBtn = document.getElementById("applyBulkCrmIdsButton");
    var countInput = document.getElementById("dmPaymentScheduleCount");
    var listEl = document.getElementById("dmScheduleList");

    if (!preview || !applyBtn) return;

    preview.addEventListener("input", function () {
      var raw = preview.textContent || "";
      var ids = parseScheduleIds(raw);
      countInput.value = ids.length || "";
    });

    preview.addEventListener("paste", function (e) {
      e.preventDefault();
      var text = (e.clipboardData || window.clipboardData).getData("text");
      document.execCommand("insertText", false, text);
    });

    applyBtn.addEventListener("click", function () {
      var raw = preview.textContent || "";
      var ids = parseScheduleIds(raw);
      if (!ids.length) {
        setStatus("dmScheduleError", "No valid schedule IDs found.", "err");
        return;
      }
      setStatus("dmScheduleError", "");
      scheduledIds = ids;
      countInput.value = ids.length;
      renderScheduleList(ids, listEl);
    });

    document.getElementById("checkingAccountButton").addEventListener("click", function () {
      document.querySelector('[name="routingNumber"]').value = "053101561";
      document.querySelector('[name="checkingAccountNumber"]').value = "7289000";
    });
  }

  function parseScheduleIds(raw) {
    return raw.split(/[\n\r\t,;]+/)
      .map(function (s) { return s.trim(); })
      .filter(function (s) { return s.length > 0; });
  }

  function renderScheduleList(ids, container) {
    container.innerHTML = "";
    ids.forEach(function (id, i) {
      var div = document.createElement("div");
      div.className = "dm-schedule-item";
      div.innerHTML =
        '<div class="dm-schedule-id-label">Schedule ' + (i+1) + '</div>' +
        '<label>ID<input name="dmScheduleId_' + i + '" value="' + esc(id) + '" readonly /></label>' +
        '<label>Amount ($)<input name="dmAmount_' + i + '" inputmode="decimal" placeholder="0.00" /></label>' +
        '<label>Due Date<span class="date-control">' +
          '<input name="dmDueDate_' + i + '" class="date-display" placeholder="MM/dd/yyyy" readonly />' +
          '<button class="date-picker-button" type="button" data-date-for="dmDueDate_' + i + '">Pick</button>' +
          '<input class="native-date-picker" type="date" data-date-picker="dmDueDate_' + i + '" tabindex="-1" aria-hidden="true" />' +
        '</span></label>' +
        '<label>Follow-up Date<span class="date-control">' +
          '<input name="dmFollowupDate_' + i + '" class="date-display" placeholder="MM/dd/yyyy" readonly />' +
          '<button class="date-picker-button" type="button" data-date-for="dmFollowupDate_' + i + '">Pick</button>' +
          '<input class="native-date-picker" type="date" data-date-picker="dmFollowupDate_' + i + '" tabindex="-1" aria-hidden="true" />' +
        '</span></label>' +
        '<label>Comment<input name="dmComment_' + i + '" /></label>';
      container.appendChild(div);
    });
    wireAllDatePickers();
  }

  function collectScheduleRows() {
    return scheduledIds.map(function (id, i) {
      return {
        dmScheduleId:   id,
        paymentAmount:  document.querySelector('[name="dmAmount_' + i + '"]')   ? document.querySelector('[name="dmAmount_' + i + '"]').value   : "",
        paymentDueDate: document.querySelector('[name="dmDueDate_' + i + '"]')  ? document.querySelector('[name="dmDueDate_' + i + '"]').value  : "",
        followupDate:   document.querySelector('[name="dmFollowupDate_' + i + '"]') ? document.querySelector('[name="dmFollowupDate_' + i + '"]').value : "",
        comment:        document.querySelector('[name="dmComment_' + i + '"]')  ? document.querySelector('[name="dmComment_' + i + '"]').value  : ""
      };
    });
  }

  function initComments() {
    document.getElementById("regenerateCommentsButton").addEventListener("click", generateComments);
    document.getElementById("escalationFollowupButton").addEventListener("click", function () {
      toggleOutcomeButton(this, "escalationFollowupButton", "escalationNoFollowupButton");
      generateComments();
    });
    document.getElementById("escalationNoFollowupButton").addEventListener("click", function () {
      toggleOutcomeButton(this, "escalationNoFollowupButton", "escalationFollowupButton");

    });
    document.querySelector('[name="paymentSetupMode"]') && document.querySelectorAll('[name="paymentSetupMode"]').forEach(function(r){
      r.addEventListener("change", updateFieldVisibility);
    });
    document.querySelector('[name="isFollowupRequired"]') && document.querySelectorAll('[name="isFollowupRequired"]').forEach(function(r){
      r.addEventListener("change", function(){
        updateFollowupState();
        generateComments();
      });
    });
  }

  function toggleOutcomeButton(clicked, clickedId, otherId) {
    clicked.classList.toggle("active");
    var other = document.getElementById(otherId);
    if (other) other.classList.remove("active");
  }

  function generateComments() {
    var data = getFormData();
    var lines = [];
    if (data.creditorName) lines.push("Creditor: " + data.creditorName);
    if (data.currentAccountNumbers) lines.push("Account #: " + data.currentAccountNumbers);
    if (scheduledIds.length) {
      var rows = collectScheduleRows();
      rows.forEach(function(r, i) {
        var line = "Schedule " + (i+1) + ": ID " + r.dmScheduleId;
        if (r.paymentAmount) line += " | Amt $" + r.paymentAmount;
        if (r.paymentDueDate) line += " | Due " + r.paymentDueDate;
        lines.push(line);
      });
    }
    if (document.getElementById("escalationFollowupButton").classList.contains("active")) {
      lines.push("Escalation/Follow up Needed");
    }
    if (document.getElementById("escalationNoFollowupButton").classList.contains("active")) {
      lines.push("Escalation/No Follow up Needed");
    }
    var ta = document.querySelector('[name="comments"]');
    if (ta) ta.value = lines.join("\n");
  }

  function updateFieldVisibility() {
    var mode = document.querySelector('[name="paymentSetupMode"]:checked');
    var modeVal = mode ? mode.value : "CBP";
    var rcLabel = document.getElementById("ringCentralMasterIdLabel");
    var repLabel = document.getElementById("repNameLabel");
    var confLabel = document.getElementById("confirmationNumberLabel");
    if (modeVal === "Web") {
      if (rcLabel) rcLabel.hidden = true;
    } else {
      if (rcLabel) rcLabel.hidden = false;
    }
  }

  function updateFollowupState() {
    var req = document.querySelector('[name="isFollowupRequired"]:checked');
    var needed = req && req.value === "Yes";
    var followupDateInput = document.getElementById("followupDate");
    var followupPicker    = document.querySelector('[data-date-for="followupDate"]');
    var followupNative    = document.querySelector('[data-date-picker="followupDate"]');
    if (followupDateInput) {
      followupDateInput.readOnly  = !needed;
      if (!needed) { followupDateInput.value = "N/A"; }
    }
    if (followupPicker)  followupPicker.disabled  = !needed;
    if (followupNative)  followupNative.disabled  = !needed;
  }

  /* ─────────────────────────────────────────────────────────
     Form submission
  ───────────────────────────────────────────────────────── */
  function initFormSubmit() {
    var form = document.getElementById("recordForm");
    if (!form) return;
    form.addEventListener("submit", function (e) {
      e.preventDefault();
      submitPaymentEntry();
    });
  }

  function getFormData() {
    var form = document.getElementById("recordForm");
    var data = {};
    new FormData(form).forEach(function (v, k) { data[k] = v; });
    return data;
  }

  function submitPaymentEntry() {
    if (!scheduledIds.length) {
      setStatus("dmScheduleError", "Apply at least one CRM Payment Schedule ID first.", "err");
      return;
    }
    var baseData = getFormData();
    var scheduleRows = collectScheduleRows();
    var contents = Object.assign({}, baseData, {
      dmScheduleMode: "Multiple",
      scheduleRows: scheduleRows,
      agentName: document.querySelector('[name="agentName"]').value,
      receivedOnDate: todayUsDate()
    });
    setSaveStatus("Saving…");
    document.querySelector('[type="submit"]').disabled = true;
    google.script.run
      .withSuccessHandler(function (res) {
        setSaveStatus(res.message || "Saved!", "ok");
        showToast(res.message || "Payment entry saved!", "ok");
        document.querySelector('[type="submit"]').disabled = false;
        if (res.ok) resetPaymentForm();
      })
      .withFailureHandler(function (err) {
        setSaveStatus(err.message, "err");
        showToast(err.message, "err", 5000);
        document.querySelector('[type="submit"]').disabled = false;
      })
      .savePaymentEntry(contents);
  }

  /* ─────────────────────────────────────────────────────────
     Export CSV
  ───────────────────────────────────────────────────────── */
  function initExportCsv() {
    document.getElementById("exportCsvButton").addEventListener("click", function () {
      var data = getFormData();
      var rows = collectScheduleRows();
      var headers = ["Client ID","Client Full Name","Creditor","Account #","Schedule ID","Amount","Due Date","Followup Date","Agent","Mode","Comments"];
      var lines = [headers.join(",")];
      rows.forEach(function (r) {
        lines.push([
          data.clientId, data.clientFullName, data.creditorName, data.currentAccountNumbers,
          r.dmScheduleId, r.paymentAmount, r.paymentDueDate, r.followupDate,
          data.agentName, data.paymentSetupMode, (data.comments || "").replace(/\n/g," ")
        ].map(function(v){ return '"' + String(v||"").replace(/"/g,'""') + '"'; }).join(","));
      });
      downloadFile("payment-entry.csv", lines.join("\n"), "text/csv");
    });
  }

  function downloadFile(name, content, type) {
    var a = document.createElement("a");
    a.href = URL.createObjectURL(new Blob([content], { type: type }));
    a.download = name;
    a.click();
  }

  function initDashboard() {
    var monthSelect = document.querySelector('[name="dashboardDueRange"]');
    populateDashboardMonths(monthSelect);

  function populateDashboardMonths(select) {
    if (!select) return;
    var d = new Date();
    select.innerHTML = "";
    for (var i = -1; i <= 3; i++) {
      var m = new Date(d.getFullYear(), d.getMonth() + i, 1);
      var label = m.toLocaleString("en-US", { month: "short", year: "numeric" });
      var val   = m.toISOString().slice(0, 7);
      var opt = document.createElement("option");
      opt.value = val;
      opt.textContent = label;
      if (i === 0) opt.selected = true;
      select.appendChild(opt);
    }
  }

  function loadDashboard() {
    var monthSelect = document.querySelector('[name="dashboardDueRange"]');
    var dueRange = monthSelect ? monthSelect.value : "";
    setStatus("dashboardStatus", "Loading dashboard…");
    document.getElementById("dashboardGrid").innerHTML = "";
    google.script.run
      .withSuccessHandler(renderDashboard)
      .withFailureHandler(function (err) {
        setStatus("dashboardStatus", err.message, "err");
      })
      .getManagementDashboardData({ dueRange: dueRange });
  }

  function renderDashboard(data) {
    setStatus("dashboardStatus", "");
    if (!data) return;
    document.getElementById("dashboardHeaderMeta").textContent = data.subtitle || "";
    var grid = document.getElementById("dashboardGrid");
    grid.innerHTML = "";
    (data.cards || []).forEach(function (card) {
      var div = document.createElement("div");
      div.className = "dashboard-card";
      var html = '<div class="dashboard-card-title">' + esc(card.title || "") + '</div>';
      if (card.type === "table") {
        html += '<table class="dashboard-table"><thead><tr>';
        (card.columns || []).forEach(function (c) { html += '<th>' + esc(c) + '</th>'; });
        html += '</tr></thead><tbody>';
        (card.rows || []).forEach(function (row) {
          html += '<tr>';
          row.forEach(function (cell) { html += '<td>' + esc(cell) + '</td>'; });
          html += '</tr>';
        });
        html += '</tbody></table>';
      } else if (card.type === "bar") {
        var max = Math.max.apply(null, (card.rows || []).map(function(r){ return Number(r[1]) || 0; })) || 1;
        (card.rows || []).forEach(function (row) {
          var pct = Math.round((Number(row[1]) || 0) / max * 100);
          html += '<div class="dashboard-bar-row">' +
            '<span class="dashboard-bar-label" title="' + esc(row[0]) + '">' + esc(row[0]) + '</span>' +
            '<div class="dashboard-bar-track"><div class="dashboard-bar-fill" style="width:' + pct + '%"></div></div>' +
            '<span class="dashboard-bar-count">' + esc(row[1]) + '</span>' +
            '</div>';
        });
      } else if (card.type === "stats") {
        (card.rows || []).forEach(function (row) {
          html += '<div class="dashboard-stat-row"><span class="dashboard-stat-label">' + esc(row[0]) +
            '</span><span class="dashboard-stat-value' + (row[2] ? ' highlight' : '') + '">' + esc(row[1]) + '</span></div>';
        });
      }
      div.innerHTML = html;
      grid.appendChild(div);
    });
    if (!data.cards || !data.cards.length) {
      grid.innerHTML = '<p style="color:var(--gray-500);padding:16px;">No dashboard data available.</p>';
    }
  }

  function exportFollowups() {
    var startFollowup = document.querySelector('[name="followupExportStartDate"]').value;
    var endFollowup   = document.querySelector('[name="followupExportEndDate"]').value;
    var startDue      = document.querySelector('[name="paymentDueExportStartDate"]').value;
    var endDue        = document.querySelector('[name="paymentDueExportEndDate"]').value;
    showToast("Loading follow-up data…");
    google.script.run
      .withSuccessHandler(function (result) {
        if (!result || !result.rows || !result.rows.length) {
          showToast("No follow-up rows found.", "err");
          return;
        }
        exportRowsToCsv("followup-report.csv", result.headers, result.rows);
        showToast("Exported " + result.rows.length + " rows.", "ok");
      })
      .withFailureHandler(function (err) { showToast(err.message, "err", 5000); })
      .getFollowupRowsBySelection({ startFollowup: startFollowup, endFollowup: endFollowup, startDue: startDue, endDue: endDue });
  }

  function buildWorkActionButtons(item) {
    if (item.status === "Completed") return '<span style="color:var(--green);font-size:12px;">&#10003; Done</span>';
    var btns = '';
    btns += '<button class="small-button" onclick="fillFormFromQueueItem(' + JSON.stringify(item).replace(/"/g,'&quot;') + ')">Fill Form</button> ';
    if (item.status !== "In Progress") {
      btns += '<button class="small-button" onclick="updateQueueItemStatus(\'' + esc(item.assignmentId) + '\',\'In Progress\')">Start</button> ';
    }
    btns += '<button class="small-button" onclick="updateQueueItemStatus(\'' + esc(item.assignmentId) + '\',\'Completed\')">Complete</button>';
    return btns;
  }

  function analyzeBulkFile() {
    var fileInput = document.querySelector('[name="bulkPaymentExcelFile"]');
    if (!fileInput.files.length) { setStatus("bulkUploadError", "Choose a file first.", "err"); return; }
    setStatus("bulkUploadError", "Parsing…");
    var file = fileInput.files[0];
    var reader = new FileReader();
    reader.onload = function (e) {
      try {
        var wb = XLSX.read(e.target.result, { type: "array", cellDates: true });
        var ws = wb.Sheets[wb.SheetNames[0]];
        var data = XLSX.utils.sheet_to_json(ws, { header: 1, raw: false, dateNF: "MM/DD/YYYY" });
        if (!data.length) { setStatus("bulkUploadError", "No data found in file.", "err"); return; }
        bulkHeaders = data[0].map(String);
        bulkParsedRows = data.slice(1).filter(function (r) { return r.some(function (v) { return v; }); });
        setStatus("bulkUploadError", "");
        renderBulkClientReview(bulkHeaders, bulkParsedRows);
      } catch (err) {
        setStatus("bulkUploadError", "Parse error: " + err.message, "err");
      }
    };
    reader.readAsArrayBuffer(file);
  }

  function renderBulkClientReview(headers, rows) {
    var clientCol    = findCol(headers, ["Client ID","ClientID","CLIENT_ID","client_id"]);
    var nameCol      = findCol(headers, ["Client Full Name","Client Name","Name","CLIENT_NAME"]);
    var scheduleCol  = findCol(headers, ["CRM Payment Schedule ID","Schedule ID","SCHEDULE_ID"]);
    var outcomeRadio = document.querySelector('[name="bulkPaymentOutcome"]:checked');
    var outcome      = outcomeRadio ? outcomeRadio.value : "Bulk Payments Setup Successfully";

    var clients = {};
    rows.forEach(function (row, i) {
      var cid = row[clientCol] || ("row-" + i);
      if (!clients[cid]) clients[cid] = { clientId: cid, name: row[nameCol] || "", schedules: [], rowIndexes: [] };
      clients[cid].schedules.push(row[scheduleCol] || "");
      clients[cid].rowIndexes.push(i);
    });

    document.getElementById("bulkClientReviewSummary").textContent = Object.keys(clients).length + " client(s), " + rows.length + " schedule(s)";
    var tbody = document.getElementById("bulkClientReviewBody");
    tbody.innerHTML = "";
    Object.values(clients).forEach(function (c) {
      var tr = document.createElement("tr");
      tr.innerHTML =
        '<td>' + esc(c.clientId) + (c.name ? '<br><small>' + esc(c.name) + '</small>' : '') + '</td>' +
        '<td>' + esc(c.schedules.length) + '</td>' +
        '<td>' + esc(outcome) + '</td>' +
        '<td><select data-client-id="' + esc(c.clientId) + '" class="bulk-client-setup">' +
          '<option value="Yes" selected>Yes</option><option value="No">No</option></select></td>' +
        '<td><input placeholder="Optional reason" data-reason-for="' + esc(c.clientId) + '" style="padding:4px 8px;border:1px solid var(--gray-300);border-radius:4px;font-size:12px;width:180px;" /></td>';
      tbody.appendChild(tr);
    });
    document.getElementById("bulkClientReviewPanel").hidden = false;
  }

  function parseNewPaymentFile() {
    var file = document.querySelector('[name="newPaymentExcelFile"]').files[0];
    if (!file) return;
    var reader = new FileReader();
    reader.onload = function (e) {
      try {
        var wb = XLSX.read(e.target.result, { type: "array", cellDates: true });
        var sheetName = document.querySelector('[name="newPaymentSheetName"]').value.trim();
        var ws = sheetName && wb.Sheets[sheetName] ? wb.Sheets[sheetName] : wb.Sheets[wb.SheetNames[0]];
        var data = XLSX.utils.sheet_to_json(ws, { header: 1, raw: false, dateNF: "MM/DD/YYYY" });
        newPaymentHeaders = data[0] || [];
        newPaymentParsedRows = data.slice(1).filter(function (r) { return r.some(function (v) { return v; }); });
        setStatus("newPaymentImportStatus", "Parsed " + newPaymentParsedRows.length + " row(s).");
      } catch (err) {
        setStatus("newPaymentImportStatus", err.message, "err");
      }
    };
    reader.readAsArrayBuffer(file);
  }

  function importNewCases() {
    if (!newPaymentParsedRows.length) { setStatus("newPaymentImportStatus", "Upload a file first.", "err"); return; }
    var sourceValues = [newPaymentHeaders].concat(newPaymentParsedRows);
    var progress = document.getElementById("newPaymentImportProgress");
    var bar      = document.getElementById("newPaymentImportProgressBar");
    var txt      = document.getElementById("newPaymentImportProgressText");
    var pct      = document.getElementById("newPaymentImportProgressPercent");
    progress.hidden = false;
    bar.style.width = "0%";
    txt.textContent = "Importing…";
    pct.textContent = "0%";
    document.getElementById("importNewCasesButton").disabled = true;

    var batchSize  = 50;
    var total      = newPaymentParsedRows.length;
    var processed  = 0;

    function runBatch() {
      var chunk = sourceValues.slice(0, batchSize + 1);
      google.script.run
        .withSuccessHandler(function (res) {
          processed += Math.min(batchSize, total - processed);
          var p = Math.round(processed / total * 100);
          bar.style.width = p + "%";
          pct.textContent = p + "%";
          if (processed >= total) {
            txt.textContent = "Import complete: " + (res.imported || 0) + " row(s) added.";
            setStatus("newPaymentImportStatus", (res.imported || 0) + " new reference rows imported.", "ok");
            document.getElementById("importNewCasesButton").disabled = false;
            showToast("Import complete!", "ok");
          } else {
            txt.textContent = "Importing… " + processed + "/" + total;
          }
        })
        .withFailureHandler(function (err) {
          setStatus("newPaymentImportStatus", err.message, "err");
          document.getElementById("importNewCasesButton").disabled = false;
          progress.hidden = true;
        })
        .importNewClientReferenceRowsFromRows(chunk);
    }
    runBatch();
  }

  /* ─────────────────────────────────────────────────────────
     CSV export helper
  ───────────────────────────────────────────────────────── */
  function exportRowsToCsv(filename, headers, rows) {
    var lines = [headers.map(csvCell).join(",")];
    rows.forEach(function (row) {
      lines.push(row.map(csvCell).join(","));
    });
    downloadFile(filename, lines.join("\n"), "text/csv");
  }

  function csvCell(v) {
    return '"' + String(v == null ? "" : v).replace(/"/g, '""') + '"';
  }

  /* ─────────────────────────────────────────────────────────
     Column finder helper
  ───────────────────────────────────────────────────────── */
  function findCol(headers, names) {
    for (var n = 0; n < names.length; n++) {
      var idx = headers.indexOf(names[n]);
      if (idx >= 0) return idx;
    }
    return 0;
  }
