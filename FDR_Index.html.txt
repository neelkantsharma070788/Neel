        return `<div class="task-row"><div class="task-id">Task ${n}</div><div class="task-name">${escapeHtml(t.name)}</div><div class="aht">${Number(t.aht).toFixed(0)} min</div><div class="task-count"><input autocomplete="off" inputmode="numeric" type="text" value="" data-task-id="${escapeHtml(t.id)}" aria-label="${escapeHtml(t.id)} count"></div><div class="task-comment"><textarea rows="1" data-task-comment="${escapeHtml(t.id)}" placeholder="Optional"></textarea></div></div>`;

      function getTaskGroups(tasks) {
        const rules = getSectionRules();
        const isAudit = (t) => /audit/i.test(t.name || "");
        const num = (t) => Number(String(t.id).replace(/\D/g, ""));
        const isOther = (t) => Boolean(rules.find((s) => s.isOther && num(t) >= s.min && num(t) <= s.max)) || /other|office hours|value added/i.test(t.name || "");
        const audit = tasks.filter(isAudit);
        const other = tasks.filter((t) => !isAudit(t) && isOther(t));
        const regular = tasks.filter((t) => !isAudit(t) && !isOther(t));
        const groups = [];
        if (rules.length) {
          rules.filter((s) => !s.isOther).forEach((s) => { const ts = regular.filter((t) => num(t) >= s.min && num(t) <= s.max); if (ts.length) groups.push({ ...s, tasks: ts }); });
        } else if (regular.length) { groups.push({ label: "Daily Tasks", color: "#064e3b", tasks: regular }); }
        if (audit.length) groups.push({ label: "Audit Tasks", color: "#7f1d1d", tasks: audit });
        if (other.length) groups.push({ label: "Other Tasks", color: "#374151", tasks: other });
        return groups;
      }

      function calculateTotals(counts) {
        const totalVolume = state.tasks.reduce((s, t) => s + (Number(counts[t.id]) || 0), 0);
        const productiveMinutes = state.tasks.reduce((s, t) => s + (Number(counts[t.id]) || 0) * t.aht, 0);
        const productiveHours = productiveMinutes / 60;
        return { totalVolume, productiveMinutes, productiveHours, utilization: productiveHours / state.workdayHours };
      }

        const t = calculateTotals(getCounts());
        const status = getStatus(t.totalVolume, t.utilization);
        el("totalVolume").textContent = formatNumber(t.totalVolume, 0);
        el("productiveMinutes").textContent = formatNumber(t.productiveMinutes, 0);
        el("productiveHours").textContent = formatNumber(t.productiveHours, 1);
        el("utilization").textContent = formatPercent(t.utilization);
        el("status").textContent = status;
        el("status").className = statusClass(status);
      }

      function submitEntry(e) {
        e.preventDefault();
        if (!el("agent").value) { showMessage("Please select an agent.", "bad"); return; }
        const payload = { date: el("workDate").value, team: state.selectedTeam, agent: el("agent").value, submittedBy: el("submittedBy").value, notes: el("notes").value, counts: getCounts(), comments: getComments() };
        if (state.localPreview) {
          const t = calculateTotals(payload.counts);
          showMessage(`Preview saved for ${state.selectedTeam}. ${formatNumber(t.totalVolume, 0)} total volume, ${formatPercent(t.utilization)} utilization.`, "good");
          showCenterToast("Entry preview saved.", "good"); clearEntryFields(); return;
        }
        el("submitButton").disabled = true;
        google.script.run
          .withSuccessHandler((r) => { showMessage(`Saved. ${formatNumber(r.totals.totalVolume, 0)} total volume, ${formatPercent(r.totals.utilization)} utilization.`, "good"); showCenterToast("Saved to Google Sheets.", "good"); clearEntryFields(); if (r.dashboard) renderDashboard(r.dashboard); el("submitButton").disabled = false; })
          .withFailureHandler((err) => { showMessage(err.message || "Could not save.", "bad"); showCenterToast(err.message || "Could not save.", "bad"); el("submitButton").disabled = false; })
          .saveDailyEntry(payload);
      }

          .withSuccessHandler((d) => { renderDashboard(d); el("refreshButton").disabled = false; })
          .withFailureHandler((e) => { showMessage(e.message || "Could not refresh dashboard.", "bad"); el("refreshButton").disabled = false; })
          .getMonthlyUtilization({ month: el("monthPicker").value, dailyDate: el("dailyDatePicker").value, team: state.selectedTeam });
      }
      function renderDashboard(dashboard) {
        const data = dashboard || emptyDashboard();
        state.currentDashboard = data;
        if (state.dashboardView === "all" && !canUseAllView()) state.dashboardView = "mine";
        const visible = applyUtilFilters(getVisibleAgentRows(data.agents || []));
        const visibleDaily = getVisibleAgentRows(data.dailyAgents || []);
        const totals = getVisibleTotals(visible);
        el("dashVolume").textContent = formatNumber(totals.totalVolume, 0);
        el("dashHours").textContent = formatNumber(totals.productiveHours, 1);
        el("dashUtilization").textContent = formatPercent(totals.avgUtilization);
        el("dashAgents").textContent = formatNumber(totals.activeAgents, 0);
        renderPie(visible);
        renderAgentRows(visible, visibleDaily);
        updateDashboardViewButtons();
      }
      function applyUtilFilters(agents) {
        let out = agents;
        if (state.agentSearch) out = out.filter((a) => a.agent.toLowerCase().includes(state.agentSearch.toLowerCase()));
        if (state.statusFilterValue) out = out.filter((a) => a.status === state.statusFilterValue);
        return out;
      }
      function getVisibleAgentRows(agents) {
        if (state.dashboardView === "all") return agents;
        const sel = el("agent").value;
        return sel ? agents.filter((a) => a.agent === sel) : [];
      }
      function getVisibleTotals(agents) {
        const activeAgents = agents.filter((a) => a.entryDays > 0).length;
        const totalVolume = agents.reduce((s, a) => s + Number(a.totalVolume || 0), 0);
        const productiveHours = agents.reduce((s, a) => s + Number(a.productiveHours || 0), 0);
        const entryDays = agents.reduce((s, a) => s + Number(a.entryDays || 0), 0);
        const avgUtilization = entryDays > 0 ? productiveHours / (entryDays * state.workdayHours) : 0;
        return { activeAgents, totalVolume, productiveHours, avgUtilization };
      }
      function renderPie(agents) {
        const active = agents.filter((a) => Number(a.productiveHours) > 0);
        const total = active.reduce((s, a) => s + Number(a.productiveHours), 0);
        if (!active.length || total === 0) { el("pieChart").className = "pie pie-empty"; el("pieChart").style.background = ""; el("pieChart").textContent = "No data"; el("pieLegend").innerHTML = ""; return; }

      function getWeekdayDates(month) {
        const [y, m] = String(month || "").split("-").map(Number);
        if (!y || !m) return [];
        const dates = []; const cur = new Date(y, m - 1, 1);
        while (cur.getMonth() === m - 1) { const day = cur.getDay(); dates.push({ date: new Date(cur), day, weekend: day === 0 || day === 6 }); cur.setDate(cur.getDate() + 1); }
        return dates;
      }

      function simulateAttendance() {
        const month = el("attMonthPicker").value || monthISO();
        const weekdays = getWeekdayDates(month).filter((d) => !d.weekend);
        const utilBy = {}; (state.currentDashboard ? state.currentDashboard.agents : []).forEach((a) => { utilBy[a.agent] = a; });
        const agents = state.agentProfiles.filter((p) => !isLeadAgentName(p.name)).map((p) => p.name);

        agents.forEach((name, idx) => {
          const u = utilBy[name];
          const days = u ? Math.min(u.entryDays || 0, weekdays.length) : Math.max(0, weekdays.length - (idx % 4) - 3);
          presentByAgent[name] = weekdays.slice(0, days).map((d) => fmtKey(d.date));
        });
        return { team: state.selectedTeam, month, agents, presentByAgent };

      function fmtKey(date) { return [date.getFullYear(), String(date.getMonth() + 1).padStart(2, "0"), String(date.getDate()).padStart(2, "0")].join("-"); }

        const dates = getWeekdayDates(month);
        const weekdays = dates.filter((d) => !d.weekend);
        const today = new Date(); today.setHours(0, 0, 0, 0);

        const presentByAgent = (res && res.presentByAgent) || {};

        const headCells = dates.map((d) => `<th class="${d.weekend ? "att-weekend" : ""}">${d.date.getDate()}</th>`).join("");

          let present = 0;
          const cells = dates.map((d) => {
            if (d.weekend) return `<td class="att-cell att-weekend"></td>`;

      function calcKraPreview() {
        const counts = getKraEntryCounts(); const m = getKraEntryMetrics();
        const totalVolume = state.tasks.reduce((s, t) => s + (Number(counts[t.id]) || 0), 0);
        const totalProduction = state.tasks.reduce((s, t) => s + (Number(counts[t.id]) || 0) * getKraWeight(t), 0);
        const target = Math.max(0, (Number(m.workingDays) || 0) - 2) * KRA_TARGET_PER_DAY;
        const productionContribution = target > 0 ? (totalProduction / target) * 0.6 : 0;
        const hasFatal = m.internalFatalErrors > 0 || m.coachingFatalErrors > 0;
        const qualityContribution = hasFatal ? 0 : m.qualityScore >= 0.99 ? 0.3 : m.qualityScore > 0.97 ? 0.24 : m.qualityScore > 0.9499 ? 0.18 : m.qualityScore > 0.92 ? 0.12 : 0;
        const shrinkageContribution = productionContribution >= 0.6 ? 0.1 : m.unplannedShrinkage >= 1 ? 0 : m.plannedShrinkage <= 2 ? 0.1 : 0;
        const overallAttainment = productionContribution + qualityContribution + shrinkageContribution;
        const rawRating = overallAttainment < 0.8 ? 1 : overallAttainment >= 1.15 ? 5 : 2 + (overallAttainment - 0.8) * (3 / 0.4);

      function getWeekdayDatesForMonth(month) {
        const [y, m] = String(month || "").split("-").map(Number);
        if (!y || !m) return [];
        const dates = []; const cur = new Date(y, m - 1, 1);
        while (cur.getMonth() === m - 1) { const day = cur.getDay(); if (day !== 0 && day !== 6) dates.push(formatDateForInput(cur)); cur.setDate(cur.getDate() + 1); }
        return dates;
      }
      function formatDateForInput(date) {
        const v = date instanceof Date ? date : new Date(date);
        if (Number.isNaN(v.getTime())) return "";
        return [v.getFullYear(), String(v.getMonth() + 1).padStart(2, "0"), String(v.getDate()).padStart(2, "0")].join("-");
      }
      function readTextFile(file) { return new Promise((res, rej) => { const r = new FileReader(); r.onload = () => res(String(r.result || "")); r.onerror = () => rej(new Error("Could not read the selected file.")); r.readAsText(file); }); }
      function readFileAsBase64(file) { return new Promise((res, rej) => { const r = new FileReader(); r.onload = () => res(String(r.result || "").split(",").pop()); r.onerror = () => rej(new Error("Could not read the selected Excel file.")); r.readAsDataURL(file); }); }
      function chooseDelimiter(text) { const first = String(text).split(/\r?\n/)[0] || ""; return (first.match(/\t/g) || []).length > (first.match(/,/g) || []).length ? "\t" : ","; }
      function parseDelimitedRows(text, delimiter) {
        const rows = []; let row = []; let cell = ""; let inQuotes = false;
        const src = String(text).replace(/\r\n/g, "\n").replace(/\r/g, "\n");
        for (let i = 0; i < src.length; i += 1) {
          const ch = src[i], next = src[i + 1];
          if (ch === '"') { if (inQuotes && next === '"') { cell += '"'; i += 1; } else { inQuotes = !inQuotes; } }
          else if (ch === delimiter && !inQuotes) { row.push(cell); cell = ""; }
          else if (ch === "\n" && !inQuotes) { row.push(cell); rows.push(row); row = []; cell = ""; }
          else { cell += ch; }
        }
        row.push(cell); rows.push(row); return rows;
      }
      function normalizeHeader(v) { return String(v || "").toLowerCase().replace(/[^a-z0-9]/g, ""); }
      function findHeaderIndex(headers, candidates) { return headers.findIndex((h) => candidates.includes(h)); }
      function parseUploadDate(value) {
        const text = String(value || "").trim();
        if (!text) return "";
        const iso = /^(\d{4})-(\d{1,2})-(\d{1,2})$/.exec(text);
        if (iso) return `${iso[1]}-${String(iso[2]).padStart(2, "0")}-${String(iso[3]).padStart(2, "0")}`;
        const slash = /^(\d{1,2})[/-](\d{1,2})[/-](\d{2,4})$/.exec(text);
        if (slash) { let f = Number(slash[1]), s = Number(slash[2]), y = Number(slash[3]); if (y < 100) y += 2000; const day = f > 12 ? f : s > 12 ? s : f; const mon = f > 12 ? s : s > 12 ? f : s; return `${y}-${String(mon).padStart(2, "0")}-${String(day).padStart(2, "0")}`; }
        if (/^\d+(\.\d+)?$/.test(text)) { const d = new Date(Math.round((Number(text) - 25569) * 86400 * 1000)); if (!Number.isNaN(d.getTime())) return formatDateForInput(d); }
        const parsed = new Date(text); return Number.isNaN(parsed.getTime()) ? "" : formatDateForInput(parsed);
      }
      function parseBulkEntries(text) {
        const delimiter = chooseDelimiter(text);
        const rows = parseDelimitedRows(text, delimiter).filter((r) => r.some((c) => String(c).trim()));
        if (rows.length < 2) throw new Error("The upload needs a header row and at least one data row.");
        const headers = rows[0].map(normalizeHeader);
        const dateIndex = findHeaderIndex(headers, ["workdate", "date"]);
        const agentIndex = findHeaderIndex(headers, ["agent", "agentname", "employeename"]);
        if (dateIndex < 0 || agentIndex < 0) throw new Error("Headers must include Work Date and Agent.");
        const notesIndex = findHeaderIndex(headers, ["generalnotes", "notes"]);
        const submittedByIndex = findHeaderIndex(headers, ["submittedby"]);
        const taskColumns = state.tasks.map((t) => ({ task: t, countIndex: findHeaderIndex(headers, [normalizeHeader(t.name), normalizeHeader(`${t.name} Count`), normalizeHeader(t.id), normalizeHeader(`${t.id} Count`)]), commentIndex: findHeaderIndex(headers, [normalizeHeader(`${t.name} Comment`), normalizeHeader(`${t.id} Comment`)]) }));
        const parsed = [];
        rows.slice(1).forEach((row, ri) => {
          const date = parseUploadDate(row[dateIndex]);
          const agent = String(row[agentIndex] || "").trim();
          if (!date && !agent) return;
          if (!date || !agent) throw new Error(`Row ${ri + 2}: Work Date and Agent are required.`);
          const counts = {}; const comments = {}; let hasData = false;
          taskColumns.forEach(({ task, countIndex, commentIndex }) => {
            const rawCount = countIndex >= 0 ? String(row[countIndex] || "").trim() : "";
            const rawComment = commentIndex >= 0 ? String(row[commentIndex] || "").trim() : "";
            const count = rawCount === "" ? "" : Number(rawCount.replace(/,/g, ""));
            if (count !== "" && (!Number.isFinite(count) || count < 0)) throw new Error(`Row ${ri + 2}: ${task.name} count must be zero or positive.`);
            counts[task.id] = count === "" ? "" : count; comments[task.id] = rawComment;
            if (count || rawComment) hasData = true;
          });
          const notes = notesIndex >= 0 ? String(row[notesIndex] || "").trim() : "";
          if (!hasData && !notes) return;
          parsed.push({ date, agent, counts, comments, notes, submittedBy: submittedByIndex >= 0 ? String(row[submittedByIndex] || "").trim() : "" });
        });
        if (!parsed.length) throw new Error("No rows with task counts or notes were found.");
        return parsed;
      }
